\cvsection{خلاصه}

مهندس نرم‌افزارم؛ پنج سال کار با \lr{Python} و تمام‌پشته با گرایش بک‌اند. در \lr{Zebracat}
سامانه‌های پشت ویدیوی مبتنی بر هوش مصنوعی را می‌سازم: خط‌لوله‌هایی که یک رندر را به کارهایی
می‌شکنند که مستقل از هم شکست می‌خورند و دوباره تلاش می‌کنند، گام‌های مدل که متن و صحنه تولید می‌کنند، ویدیوی برنامه‌پذیر با \lr{Remotion}، و رابط‌های \lr{React} روی همهٔ این‌ها. دربارهٔ رفتار واقعی مدل‌های زبانی زیر بار کاربر می‌نویسم.

\cvsection{تجربهٔ کاری}

\role{مهندس نرم‌افزار \textperiodcentered\ هوش مصنوعی}{\lr{Zebracat}}{پلتفرم ساخت ویدیو از متن}{شهریور ۱۴۰۳ — اکنون · دورکاری (برلین)}
\begin{cvbullets}
  \item ساخت خط‌لولهٔ تولید: هر رندر به کارهایی شکسته می‌شود که مستقل از هم روی کارگرهای
        \lr{RabbitMQ} دوباره تلاش می‌کنند و وضعیت هر کار جداگانه نگه داشته می‌شود، تا شکست یک
        مرحله به قیمت کل ویدیو تمام نشود.
  \item کار روی بخش مدلِ محصول — ساخت متن و صحنه، اعتبارسنجی خروجی با شِما پیش از رسیدن به
        خط‌لوله، و تلاش مجددی که نمی‌گذارد یک پاسخ نامعتبر کل رندر را خراب کند.
  \item ساخت ویدیوی برنامه‌پذیر با \lr{Remotion}: کامپوننت‌های \lr{React} که بدون مرورگر به فریم
        تبدیل می‌شوند، تا تغییر یک قالب مثل هر تغییر کد دیگری منتشر شود.
  \item کار در همهٔ لایه‌ها — سرویس‌های \lr{Django} و \lr{FastAPI} پشت رابط‌های \lr{React} و
        \lr{TypeScript}.
\end{cvbullets}
\tech{Python · FastAPI · Django · RabbitMQ · Redis · React · TypeScript · Remotion · Docker · AWS}

\vspace{6pt}

\role{توسعه‌دهندهٔ تمام‌پشته}{فریلنس}{محصولات وب و \lr{SaaS}}{خرداد ۱۴۰۰ — شهریور ۱۴۰۳ · دورکاری}
\begin{cvbullets}
  \item تحویل کامل محصول به کارفرما: مدل داده، \lr{API}، رابط کاربری، استقرار، و پشتیبانی پس از آن.
  \item استانداردسازی روی \lr{Django/DRF} و \lr{FastAPI} پشت \lr{React}، کانتینری‌شده و با پایش
        فعال پیش از تحویل.
  \item برعهده‌گرفتن بخش‌هایی که کارفرما خودش نمی‌توانست بررسی‌شان کند — احراز هویت، کنترل
        دسترسی، و نحوهٔ نگهداری داده‌هایش.
\end{cvbullets}
\tech{Django · DRF · FastAPI · React · PostgreSQL · Docker · Nginx}
