\cvsection{خلاصه}

مهندس نرم‌افزارم؛ پنج سال \lr{Python}، بیشتر در بک‌اند. در \lr{Zebracat} روی مسیر متن تا ویدیوی
نهایی کار می‌کنم: کارهایی که خودشان دوباره تلاش می‌کنند، خروجی مدلی که باید از شِما رد شود، و
قالب‌های \lr{Remotion} که مثل کد منتشر می‌شوند. درباره‌ی سامانه‌های مبتنی بر مدل زبانی می‌نویسم،
وقتی کاربر واقعی به آن‌ها می‌رسد.

\cvsection{تجربهٔ کاری}

\role{مهندس نرم‌افزار \textperiodcentered\ هوش مصنوعی}{\lr{Zebracat}}{پلتفرم ساخت ویدیو از متن}{شهریور ۱۴۰۳ — اکنون · دورکاری (برلین)}
\begin{cvbullets}
  \item دیگر شکست یک صحنه کل رندر را نمی‌کشد. هر مرحله یک کار روی \lr{RabbitMQ} است، با وضعیت و
        تلاش مجدد خودش.
  \item متن و صحنه تولید می‌شوند، بعد با شِما سنجیده می‌شوند. پاسخ خراب دوباره تلاش می‌کند؛
        ویدیو را پایین نمی‌آورد.
  \item قالب‌ها \lr{Remotion} هستند — \lr{React} که بدون مرورگر به فریم تبدیل می‌شود. تغییر یکی
        یعنی یک تغییر کد.
  \item سرویس‌های \lr{Django} و \lr{FastAPI} پشت \lr{React} و \lr{TypeScript}.
\end{cvbullets}
\tech{Python · FastAPI · Django · RabbitMQ · Redis · React · TypeScript · Remotion · Docker · AWS}

\vspace{6pt}

\role{توسعه‌دهندهٔ تمام‌پشته}{فریلنس}{محصولات وب و \lr{SaaS}}{خرداد ۱۴۰۰ — شهریور ۱۴۰۳ · دورکاری}
\begin{cvbullets}
  \item محصول کارفرما را از مدل داده تا \lr{API} و رابط و استقرار بردم، و بعد از تحویل هم ماندم.
  \item \lr{Django/DRF} یا \lr{FastAPI} پشت \lr{React}، کانتینری‌شده، با پایش فعال پیش از تحویل.
  \item احراز هویت، کنترل دسترسی، و نحوهٔ نگهداری داده — بخش‌هایی که کارفرما خودش نمی‌توانست
        بررسی‌شان کند.
\end{cvbullets}
\tech{Django · DRF · FastAPI · React · PostgreSQL · Docker · Nginx}
