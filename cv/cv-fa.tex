\cvsection{خلاصه}

مهندس نرم‌افزارم با پنج سال تجربهٔ \lr{Python}، بیشتر در بک‌اند، و تمام‌پشته هر جا کار لازم داشته باشد.
در \lr{Zebracat}، شرکتی که با هوش مصنوعی ویدیو می‌سازد، کار می‌کنم و تقریباً هر هفته در \lr{Medium} دربارهٔ
مهندسی هوش مصنوعی می‌نویسم: اینکه \lr{RAG}، پرامپت، ارزیابی و انتخاب مدل در محیط واقعی چطور از پس کار برمی‌آیند.

\cvsection{تجربهٔ کاری}

\role{مهندس نرم‌افزار، هوش مصنوعی}{\lr{Zebracat}}{پلتفرم ساخت ویدیو با هوش مصنوعی}{شهریور ۱۴۰۳ — اکنون · دورکاری}
\begin{cvbullets}
  \item قابلیت‌های محصول را از سر تا ته می‌سازم، از سرویس‌های بک‌اند تا وب‌اپی که کاربران هر روز با آن کار می‌کنند.
  \item توانایی‌های تازهٔ هوش مصنوعی را به قابلیت‌هایی تبدیل می‌کنم که در برخورد با کاربر واقعی هم سرپا بمانند.
  \item نیمهٔ کم‌سروصدای کار برایم مهم است: پایداری، رابط‌های روشن، و کدی که نفر بعدی بتواند تغییرش بدهد.
\end{cvbullets}

\vspace{6pt}

\role{توسعه‌دهندهٔ تمام‌پشته}{فریلنس}{محصولات وب و \lr{SaaS}}{خرداد ۱۴۰۰ — شهریور ۱۴۰۳ · دورکاری}
\begin{cvbullets}
  \item محصول کارفرما را از مخزن خالی تا محیط عملیاتی بردم: مدل داده، \lr{API}، رابط کاربری، استقرار؛ و بعد هم برای پشتیبانی ماندم.
  \item مسئول بخش‌هایی بودم که کارفرما به‌راحتی نمی‌تواند خودش بررسی کند: احراز هویت، کنترل دسترسی، و نحوهٔ نگهداری داده‌ها.
  \item پروژه را با پایش و تست آماده تحویل دادم تا توسعه‌دهندهٔ بعدی کورکورانه شروع نکند.
\end{cvbullets}
